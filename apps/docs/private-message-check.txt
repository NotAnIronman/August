// Run in the logged-in game tab's browser Console, with a private message typed
// but not submitted. This reads diagnostic state only. It does NOT send a message,
// change settings, or print message text, recipient names, passwords or tokens.
(async () => {
    const client = window.__osrsClient || Array.from(document.querySelectorAll('canvas'))
        .map(canvas => canvas.__osrsClient).find(Boolean);
    const scriptUrl = Array.from(document.scripts, script => script.src)
        .find(src => /\/(?:main[.\-]|bundle\.js)/.test(src) && new URL(src, location.href).origin === location.origin);
    const report = { clientAvailable: !!client, bundleFile: scriptUrl ? new URL(scriptUrl).pathname : null };
    if (scriptUrl) {
        try {
            const response = await fetch(scriptUrl, { cache: 'no-store' });
            const source = response.ok ? await response.text() : '';
            Object.assign(report, {
                bundleStatus: response.status,
                clientHookInBundle: source.includes('__osrsClient'),
                privatePromptBridgeInBundle: source.includes('consumeSubmittedPrivateMessage'),
                currentMenuCleanupInBundle: source.includes('releaseContextMenuWidgetInput'),
            });
        } catch { report.bundleReadFailed = true; }
    }
    if (client?.varManager) {
        const vars = client.varManager;
        report.privateInput = {
            mode: vars.getVarcInt(5),
            blocked: vars.getVarbit(4394),
            length: vars.getVarcString(359)?.length ?? 0,
            hasRecipient: !!vars.getVarcString(360),
            dialog: client.cs2Vm?.inputDialogType,
            privateFilter: client.cs2Vm?.privateChatMode,
        };
    }
    console.log('[August PM check]', JSON.stringify(report));
})();
